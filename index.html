PHP 8.4 
namespace App\Domain\Education\Actions;

final readonly class AccessibleStudyAction
{
    public function execute(Developer $student, Module $pdf): StudySession
    {
        $pdf->renderDocument(format: 'PDF/A');
        $pdf->enableVoiceReader(lang: 'pt-BR');

        return new StudySession(cost: 0.00, accessible: true);
    }
}
